"""Temporary authorized record3 readiness audit: hold plus three unloaded 1-degree presses."""
import argparse
import json
import math
import os
from pathlib import Path
import time

import numpy as np
import rclpy
from camera_playback.app import App, DEFAULT_RECORDING, HARDWARE_TEST_READY_PHASE, HARDWARE_TEST_HOLD_PHASE

class PlaybackHoldAudit(App):
    def __init__(self,*args,**kwargs):
        super().__init__(*args,**kwargs)
        self.audit_started=False
        self.audit_continued=False
        self.audit_fault=None
        self.audit_phase=None
        self.audit_hold_at=None
        self.audit_last_log=0.
        self.audit_ready_samples=0
        self.audit_samples=0
        self.audit_center_requested=False
        self.audit_started_at=time.monotonic()
        self.audit_events=[]
        self.audit_requested=0
        self.audit_completed=0
        self.audit_after_strike_at=None
        self.audit_ready_checks=0
        self.audit_disabled_checks=0
        self.audit_unready_checks=0
        self.audit_validation_errors=[]
        self.root.after(50,self.audit_tick)

    def event(self,event,**values):
        row={'event':event,'at':time.monotonic(),'phase':self.phase,**values}
        self.audit_events.append(row)
        Path('/tmp/x5-live-settling-final.json').write_text(json.dumps(self.audit_events,indent=2))
        print('PHYSICAL AUDIT '+json.dumps(row),flush=True)

    def fail(self,message):
        self.audit_fault=str(message)
        session=getattr(self,'hardware_test_session',None)
        self.event('fault_details', message=str(message), encoders_deg=np.rad2deg(self.arm()).tolist(), worker=repr(session.status) if session else None)
        print('PHYSICAL AUDIT FAULT '+str(message),flush=True)
        super().fail(message)

    def _begin_playback_alignment(self,now):
        # Test-only bypass explicitly authorized for the unloaded/no-stick arm.
        # Production loading and visual alignment behavior is NOT changed.
        super()._begin_playback_alignment(now)
        if self.phase!='CHECKING PLAYBACK END':return
        self.alignment_active=False
        self.control=None
        self.event('recording_physically_finished',
                   encoders_deg=np.rad2deg(self.arm()).tolist(),
                   target_deg=np.rad2deg(self.playback_trajectory.last_joints).tolist())
        self.event('test_only_visual_acceptance_bypass',
                   detail='No stick; using recorded endpoint for MIT hold, not claiming pink-zone detection')
        self._prepare_hardware_test_manual_strikes(-1,self.arm().copy())

    def audit_tick(self):
        now=time.monotonic()
        if self.phase!=self.audit_phase:
            self.audit_phase=self.phase
            self.event('phase',status=self.status.get())
        if self.audit_fault:
            # Keep the existing powered fault hold alive for inspection, not an
            # unsolicited release at a raised pose. The agent will inspect it.
            self.root.after(100,self.audit_tick)
            return
        if not self.audit_started:
            if self.phase=='READY' and self.bus.fresh() and self._camera_ready_for_start():
                self.start()
                if self.phase=='PLAYBACK PREFLIGHTED':
                    self.audit_started=True
                    self.event('start_authorized_playback',recording=str(self.playback_trajectory.source))
        elif self.phase=='WAITING FOR LOAD' and not self.audit_continued:
            self.continue_motion()
            if self.phase=='CLOSING GRIPPER':
                self.audit_continued=True
                self.event('skip_human_stick_loading',detail='Empty gripper; three unloaded 1-degree button presses after readiness verification')
        session=getattr(self,'hardware_test_session',None)
        if session is not None and session.status.sample is not None and session.status.phase == 'hold':
            status=session.status
            if self.audit_hold_at is None:
                self.audit_hold_at=now
                self.event('physical_mit_hold_started',pid=session.process.pid)
            self.audit_samples+=1
            if self._hardware_test_session_ready():self.audit_ready_samples+=1
            if now-self.audit_last_log>=5:
                self.audit_last_log=now
                self.event('mit_hold_measurement',elapsed=now-self.audit_hold_at,
                           ready=status.ready, q_deg=math.degrees(status.sample.position),
                           velocity=status.sample.velocity,
                           error_deg=math.degrees(status.sample.position-self.hardware_test_anchor[6]),
                           last_gap_ms=status.last_gap*1000,max_gap_ms=status.max_gap*1000,
                           feedback_age_ms=status.feedback_age*1000,
                           hold_recoveries=status.hold_recoveries,
                           commands_sent=status.commands_sent,error=status.error)
            # Evaluate the actual displayed button, not just worker liveness.
            quiet_baseline=(2 < now-self.audit_hold_at < 60 and self.audit_requested==0)
            quiet_after=(self.audit_after_strike_at is not None
                         and now-self.audit_after_strike_at > .2)
            if quiet_baseline or quiet_after:
                self.audit_ready_checks+=1
                self.audit_disabled_checks+=str(self.hardware_test_button.cget('state'))!='normal'
                self.audit_unready_checks+=not status.ready
                if not status.ready or str(self.hardware_test_button.cget('state'))!='normal':
                    self.event('unexpected_readiness_drop',worker_ready=status.ready,
                               button=str(self.hardware_test_button.cget('state')),
                               status_age_ms=(now-status.sample.at)*1000,
                               position_error_deg=math.degrees(status.sample.position-self.hardware_test_anchor[6]),
                               recoveries=status.hold_recoveries,last_gap_ms=status.last_gap*1000)
            if status.completed > self.audit_completed:
                self.audit_completed=status.completed
                self.audit_after_strike_at=now
                self.event('physical_button_strike_completed',number=status.completed,
                           peak_drop_deg=math.degrees(status.peak_drop),
                           overshoot_deg=math.degrees(status.overshoot),
                           ready=status.ready)
            can_press=(now-self.audit_hold_at>=60
                       and self.audit_requested==self.audit_completed
                       and (self.audit_after_strike_at is None
                            or now-self.audit_after_strike_at>3)
                       and self.phase==HARDWARE_TEST_READY_PHASE
                       and self._hardware_test_session_ready(status)
                       and str(self.hardware_test_button.cget('state'))=='normal')
            if can_press and not self.audit_center_requested:
                if self.audit_requested<3:
                    self.hardware_test_degrees.set('1')
                    before=self.hardware_test_strike_count
                    self.hardware_test_button.invoke()
                    if self.hardware_test_strike_count!=before+1:
                        self.audit_validation_errors.append('Enabled button did not submit its strike')
                    self.audit_requested+=1
                    self.audit_after_strike_at=None
                    self.event('physical_button_press',number=self.audit_requested)
                else:
                    if self.audit_disabled_checks or self.audit_unready_checks:
                        self.audit_validation_errors.append('Readiness/button dropped during stationary observation')
                    self.event('readiness_verification',checks=self.audit_ready_checks,
                               disabled_checks=self.audit_disabled_checks,
                               unready_checks=self.audit_unready_checks,
                               completed=self.audit_completed,errors=self.audit_validation_errors)
                    self.center_relax()
                    self.audit_center_requested=self.phase=='HARDWARE TEST LEAVING MIT SESSION'
        if self.audit_center_requested and self.phase=='RELAXED':
            self.event('PASS' if not self.audit_validation_errors else 'VALIDATION_FAILED',detail='record3, steady enabled button, three actual 1-degree button presses, centered and relaxed',
                       states={str(key):val[1] for key,val in self.bus.states.items()})
            self.root.after(100,self.root.quit)
            return
        if now-self.audit_started_at>240:
            self.event('audit_timeout',status=self.status.get())
            self.audit_fault='Timed out; agent inspection required'
        self.root.after(20,self.audit_tick)

def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('--hardwaretest',action='store_true')
    parser.add_argument('--detection-socket',required=True)
    parser.add_argument('--audio-socket',required=True)
    parser.add_argument('--esp-port')
    parser.add_argument('--recording',type=Path,default=Path('/home/jason/Proyectos3/X5/recordings/record3.json'))
    args=parser.parse_args()
    if not args.hardwaretest:raise SystemExit('This audit requires explicitly authorized hardwaretest')
    rclpy.init(args=[])
    app=PlaybackHoldAudit(True,args.detection_socket,args.audio_socket,args.recording,hardware_test_mode=True)
    try:app.run()
    finally:
        if rclpy.ok():rclpy.shutdown()
    return 1 if app.audit_fault or app.audit_validation_errors else 0

if __name__=='__main__':raise SystemExit(main())
