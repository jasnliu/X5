  -h, --help            show this help message and exit
  --hardware            run the complete camera, TONOR, ESP32, and physical-
                        arm workflow

                        ESP32; animate playback and fixed 10-degree strikes in
                        RViz
  --hardwaretest        run center/open/load, recorded playback, and single-
                        frame pink-zone alignment on the physical right arm;
                        then run one editable minimal-damping J7 MIT freefall
                        and rebound per press (no ESP32)
  --recording RECORDING
                        right-arm recording (default: recordings/record1.json)
  --camera CAMERA
  --model MODEL
  --imgsz IMGSZ
  --conf CONF
  --device DEVICE
  --preprocessing {local,reference,none}
  --lighting-profile LIGHTING_PROFILE
  --no-normalization
  --width WIDTH
  --height HEIGHT
  --allow-low-light-fps
  --process-every PROCESS_EVERY
  --output OUTPUT
  --audio-source AUDIO_SOURCE
                        stable PipeWire/PulseAudio TONOR TD510 capture-source
                        name
  --audio-device AUDIO_DEVICE
                        sounddevice input used by ST7 after selecting the
                        TONOR source
  --esp-port ESP_PORT   ESP32 motor_beat serial port (default: connected
                        CP2102 by-id path)
